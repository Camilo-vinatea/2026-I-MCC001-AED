34template <typename Container, typename ObjType>
35btree_pos_t binary_search(Container& container, btree_pos_t first, btree_pos_t last, ObjType &object)
36{
37 if( first >= last )
38 return first;
39 while( first < last )
40 {
41 btree_pos_t mid = (first+last)/2;
42 if( object == (ObjType)container[mid] )
43 return mid;
44 if( object > (ObjType)container[mid] )
45 first = mid+1;
46 else
47 last = mid;
48 }
49 if( object <= (ObjType)container[first] )
50 return first;
51 return last;
52}

85struct BTreeForwardInorderPolicy {
87 template <typename Page, typename Node>
88 static void construir(std::deque<std::pair<Node*, tree_height_t>>& cola,
89 Page* p, tree_height_t nivel = 0) {
90 if (!p) return;
91 for (btree_pos_t i = 0; i < p->m_KeyCount; ++i) {
92 construir<Page, Node>(cola, p->m_SubPages[i], nivel + 1);
93 cola.push_back({&p->m_Keys[i], nivel});
94 }
95 construir<Page, Node>(cola, p->m_SubPages[p->m_KeyCount], nivel + 1);
96 }
97};

106 BTreeForwardInorderPolicy::construir<Page, Node>(cola, p, nivel);

331template <typename Traits>
332bt_ErrorCode CBTreePage<Traits>::Insert(const keyType& key, const ObjIDType ObjID)
333{
334 btree_pos_t pos = binary_search(m_Keys, 0, m_KeyCount, key);
335 bt_ErrorCode error = bt_ok;
336
337 if( pos < m_KeyCount && (keyType)m_Keys[pos] == key && m_Unique)
338 return bt_duplicate;
339
340 if( !m_SubPages[pos] ) // hoja
341 {
342 ::insert_at(m_Keys, Node(key, ObjID), pos);
343 NumberOfKeys()++;
344 if( Overflow() )
345 return bt_overflow;
346 return bt_ok;
347 }
348 // recursión + propagación de overflow
349 error = m_SubPages[pos]->Insert(key, ObjID);
350 if( error == bt_overflow )
351 {
352 if( !Redistribute1(pos) )
353 SplitChild(pos);
354 if( Overflow() )
355 return bt_overflow;
356 return bt_ok;
357 }
358 return bt_ok;
359}

363template <typename Traits>
364bool CBTreePage<Traits>::Redistribute1(btree_pos_t &pos)
365{
366 if( m_SubPages[pos]->Underflow() )
367 {
368 btree_pos_t nkol = 0, nkor = 0;
369 if( pos > 0 ) nkol = m_SubPages[pos-1]->NumberOfKeys();
370 if( pos < NumberOfKeys() ) nkor = m_SubPages[pos+1]->NumberOfKeys();
371
372 if( nkol > nkor ){
373 if( m_SubPages[pos-1]->NumberOfKeys() > m_SubPages[pos-1]->MinNumberOfKeys() )
374 RedistributeL2R(pos-1);
375 else {
376 if( pos == NumberOfKeys() ) --pos;
377 return false;
378 }
379 } else {
380 if( m_SubPages[pos+1]->NumberOfKeys() > m_SubPages[pos+1]->MinNumberOfKeys() )
381 RedistributeR2L(pos+1);
382 else {
383 if( pos == 0 ) ++pos;
384 return false;
385 }
386 }
387 }
388 else // overflow
389 {
390 btree_pos_t fcol = GetFreeCellsOnLeft(pos);
391 btree_pos_t fcor = GetFreeCellsOnRight(pos);
392
393 if( !fcol && !fcor && m_SubPages[pos]->IsFull() )
394 return false;
395 if( fcol > fcor )
396 RedistributeR2L(pos);
397 else
398 RedistributeL2R(pos);
399 }
400 return true;
401}

404template <typename Traits>
405bool CBTreePage<Traits>::Redistribute2(btree_pos_t pos)
406{
407 assert( pos > 0 && pos < NumberOfKeys() );
408 assert( m_SubPages[pos-1] != 0 && m_SubPages[pos] != 0 && m_SubPages[pos+1] != 0 );
409 assert( m_SubPages[pos-1]->Underflow() ||
410 m_SubPages[ pos ]->Underflow() ||
411 m_SubPages[pos+1]->Underflow() );
412
413 if( m_SubPages[pos-1]->Underflow() )
414 {
415 RedistributeR2L(pos+1);
416 RedistributeR2L(pos);
417 if( m_SubPages[pos-1]->Underflow() ) return false;
418 }
419 else if( m_SubPages[pos+1]->Underflow() )
420 {
421 RedistributeL2R(pos-1);
422 RedistributeL2R(pos);
423 if( m_SubPages[pos+1]->Underflow() ) return false;
424 }
425 else
426 {
427 RedistributeL2R(pos-1);
428 RedistributeR2L(pos+1);
429 if( m_SubPages[pos]->Underflow() ) return false;
430 }
431 return true;
432}

435template <typename Traits>
436void CBTreePage<Traits>::RedistributeR2L(btree_pos_t pos)
437{
438 BTPage *pSource = m_SubPages[pos],
439 *pTarget = m_SubPages[pos-1];
440
441 while(pSource->GetNumberOfKeys() > pSource->MinNumberOfKeys() &&
442 pTarget->GetNumberOfKeys() < pSource->GetNumberOfKeys() )
443 {
444 ::insert_at(pTarget->m_Keys, m_Keys[pos-1], pTarget->NumberOfKeys()++);
445 ::insert_at(pTarget->m_SubPages, pSource->m_SubPages[0], pTarget->NumberOfKeys());
446 m_Keys[pos-1] = pSource->m_Keys[0];
447 ::remove(pSource->m_Keys , 0);
448 ::remove(pSource->m_SubPages, 0);
449 pSource->NumberOfKeys()--;
450 }

471template <typename Traits>
472void CBTreePage<Traits>::SplitChild(btree_pos_t pos)
473{
474 BTPage *pChild1 = 0, *pChild2 = 0;
475 if( pos > 0 )
476 if( m_SubPages[pos-1]->IsFull() )
477 { pChild1 = m_SubPages[pos-1]; pChild2 = m_SubPages[pos--]; }
478 if( pos < GetNumberOfKeys() )
479 if( m_SubPages[pos+1]->IsFull() )
480 { pChild1 = m_SubPages[pos]; pChild2 = m_SubPages[pos+1]; }
481
482 vector<Node> tmpKeys;
483 vector<BTPage *> tmpSubPages;
484
485 MovePage(pChild1, tmpKeys, tmpSubPages);
486 tmpKeys.push_back(m_Keys[pos]);
487 MovePage(pChild2, tmpKeys, tmpSubPages);
488
489 BTPage *pChild3 = 0;
490 Node oi1, oi2;
491 SplitPageInto3(tmpKeys, tmpSubPages, pChild1, pChild2, pChild3, oi1, oi2);
492
493 m_Keys[pos] = oi1;
494 m_SubPages[pos] = pChild1;
495 ::insert_at(m_Keys, oi2, pos+1);
496 ::insert_at(m_SubPages, pChild2, pos+1);
497 NumberOfKeys()++;
498
499 m_SubPages[pos+2] = pChild3;
500}

504template <typename Traits>
505void CBTreePage<Traits>::SplitPageInto3(vector<Node>& tmpKeys,
506 vector<BTPage *> & tmpSubPages,
507 BTPage* & pChild1,
508 BTPage* & pChild2,
509 BTPage* & pChild3,
510 Node & oi1,
511 Node & oi2)
512{
513 assert(tmpKeys.size() >= 8);
514 assert(tmpSubPages.size() >= 9);
515 if( !pChild1 )
516 pChild1 = new BTPage(m_MaxKeysForChilds, m_Unique);
517
518 pChild1->clear();
519 btree_pos_t nKeys = (tmpKeys.size()-2)/3;
520 btree_pos_t i = 0;
521 for( ; i < nKeys; ++i ) {
522 pChild1->m_Keys[i] = tmpKeys[i];
523 pChild1->m_SubPages[i] = tmpSubPages[i];
524 pChild1->NumberOfKeys()++;
525 }
526 pChild1->m_SubPages[i] = tmpSubPages[i];
527
528 oi1 = tmpKeys[i++];
529
530 if( !pChild2 )
531 pChild2 = new BTPage(m_MaxKeysForChilds, m_Unique);
532 pChild2->clear();
533 nKeys += (tmpKeys.size()-2)/3 + 1;
534 btree_pos_t j = 0;
535 for(; i < nKeys; ++i, ++j ) {
536 pChild2->m_Keys[j] = tmpKeys[i];
537 pChild2->m_SubPages[j] = tmpSubPages[i];
538 pChild2->NumberOfKeys()++;
539 }
540 pChild2->m_SubPages[j] = tmpSubPages[i];
541
542 oi2 = tmpKeys[i++];
543
544 if( !pChild3 )
545 pChild3 = new BTPage(m_MaxKeysForChilds, m_Unique);
546 pChild3->clear();
547 nKeys = (btree_pos_t)tmpKeys.size();
548 for(j = 0; i < nKeys; ++i, ++j) {
549 pChild3->m_Keys[j] = tmpKeys[i];
550 pChild3->m_SubPages[j] = tmpSubPages[i];
551 pChild3->NumberOfKeys()++;
552 }
553 pChild3->m_SubPages[j] = tmpSubPages[i];
554}

578template <typename Traits>
579bool CBTreePage<Traits>::Search(const keyType &key, obj_id_t &ObjID)
580{
581 btree_pos_t pos = binary_search(m_Keys, 0, m_KeyCount, key);
582 if( pos >= m_KeyCount ){
583 if( m_SubPages[pos] )
584 return m_SubPages[pos]->Search(key, ObjID);
585 else
586 return false;
587 }
588 if( key == m_Keys[pos].key ) {
589 ObjID = m_Keys[pos].ObjID;
590 m_Keys[pos].UseCounter++;
591 return true;
592 }
593 if( key < m_Keys[pos].key )
594 if( m_SubPages[pos] )
595 return m_SubPages[pos]->Search(key, ObjID);
596 return false;
597}

604template <typename Traits>

617 else if (func(*it, it.level(), std::forward<Args>(args)...))

619 }
620 return nullptr;

641template <typename Traits>
642bt_ErrorCode CBTreePage<Traits>::Remove(const keyType &key, const ObjIDType ObjID)
643{
644 bt_ErrorCode error = bt_ok;
645 btree_pos_t pos = binary_search(m_Keys, 0, m_KeyCount, key);
646 if( pos < NumberOfKeys() && key == m_Keys[pos].key )
647 {
648 if( !m_SubPages[pos+1] )
649 {
650 ::remove(m_Keys, pos);
651 NumberOfKeys()--;
652 if( Underflow() ) return bt_underflow;
653 return bt_ok;
654 }
655 {
656 Node &rFirstFromRight = m_SubPages[pos+1]->GetFirstNode();
657 swap(m_Keys[pos], rFirstFromRight);
658 error = m_SubPages[++pos]->Remove(key, ObjID);
659 }
660 }
661 else if( pos == NumberOfKeys() )
662 error = m_SubPages[pos]->Remove(key, ObjID);
663 else if( key <= m_Keys[pos].key ){
664 if( m_SubPages[pos] )
665 error = m_SubPages[pos]->Remove(key, ObjID);
666 else
667 return bt_nofound;
668 }
669 if( error == bt_underflow ){
670 if( TreatUnderflow(pos) )
671 return bt_ok;
672 if( IsRoot() && NumberOfKeys() == 2 )
673 return MergeRoot();
674 return Merge(pos);
675 }
676 if( error == bt_nofound )
677 return bt_nofound;
678 return bt_ok;
679}

682template <typename Traits>
683bt_ErrorCode CBTreePage<Traits>::Merge(btree_pos_t pos)
684{
685 assert( m_SubPages[pos-1]->NumberOfKeys() +
686 m_SubPages[ pos ]->NumberOfKeys() +
687 m_SubPages[pos+1]->NumberOfKeys() ==
688 3*m_SubPages[ pos ]->MinNumberOfKeys() - 1);
689
690 vector<Node> tmpKeys;
691 vector<BTPage *> tmpSubPages;
692
693 BTPage *pChild1 = m_SubPages[pos-1],
694 *pChild2 = m_SubPages[ pos ],
695 *pChild3 = m_SubPages[pos+1];
696 MovePage(pChild1, tmpKeys, tmpSubPages);
697 tmpKeys.push_back(m_Keys[pos-1]);
698 MovePage(pChild2, tmpKeys, tmpSubPages);
699 tmpKeys.push_back(m_Keys[ pos ]);
700 MovePage(pChild3, tmpKeys, tmpSubPages);
701 pChild3->Destroy();
702
703 btree_pos_t nKeys = pChild1->GetFreeCells();
704 btree_pos_t i = 0;
705 for( ; i < nKeys ; ++i ) {
706 pChild1->m_Keys[i] = tmpKeys[i];
707 pChild1->m_SubPages[i] = tmpSubPages[i];
708 pChild1->NumberOfKeys()++;
709 }
710 pChild1->m_SubPages[i] = tmpSubPages[i];
711
712 m_Keys[pos-1] = tmpKeys[i];
713 m_SubPages[pos-1] = pChild1;
714
715 ::remove(m_Keys, pos);
716 ::remove(m_SubPages, pos);
717 NumberOfKeys()--;
718
719 nKeys = pChild2->GetFreeCells();
720 btree_pos_t j = ++i;
721 for(i = 0 ; i < nKeys ; ++i, ++j ) {
722 pChild2->m_Keys[i] = tmpKeys[j];
723 pChild2->m_SubPages[i] = tmpSubPages[j];
724 pChild2->NumberOfKeys()++;
725 }
726 pChild2->m_SubPages[i] = tmpSubPages[j];
727 m_SubPages[pos] = pChild2;
728
729 if( Underflow() ) return bt_underflow;
730 return bt_ok;
731}

734template <typename Traits>
735bt_ErrorCode CBTreePage<Traits>::MergeRoot()
736{
737 btree_pos_t pos = 1;
738 assert( m_SubPages[pos-1]->NumberOfKeys() +
739 m_SubPages[ pos ]->NumberOfKeys() +
740 m_SubPages[pos+1]->NumberOfKeys() ==
741 3*m_SubPages[ pos ]->MinNumberOfKeys() - 1);
742
743 BTPage *pChild1 = m_SubPages[pos-1],
744 *pChild2 = m_SubPages[ pos ],
745 *pChild3 = m_SubPages[pos+1];
746 btree_pos_t nKeys = pChild1->NumberOfKeys() + pChild2->NumberOfKeys() + pChild3->NumberOfKeys() + 2;
747
748 vector<Node> tmpKeys;
749 vector<BTPage *> tmpSubPages;
750
751 MovePage(pChild1, tmpKeys, tmpSubPages);
752 tmpKeys.push_back(m_Keys[pos-1]);
753 MovePage(pChild2, tmpKeys, tmpSubPages);
754 tmpKeys.push_back(m_Keys[ pos ]);
755 MovePage(pChild3, tmpKeys, tmpSubPages);
756
757 clear();
758 btree_pos_t i = 0;
759 for( ; i < nKeys ; ++i ){
760 m_Keys[i] = tmpKeys[i];
761 m_SubPages[i] = tmpSubPages[i];
762 NumberOfKeys()++;
763 }
764 m_SubPages[i] = tmpSubPages[i];
765
766 pChild1->Destroy();
767 pChild2->Destroy();
768 pChild3->Destroy();
769
770 return bt_rootmerged;
771}

784template <typename Traits>
785void CBTreePage<Traits>::Create()
786{
787 Reset();
788 m_Keys.resize(m_MaxKeys+1);

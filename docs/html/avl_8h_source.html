35template <typename Traits>
36class AVL : public BinaryTree<Traits> {
37 using Base = BinaryTree<Traits>;
38 using MyNode = AVLNode<Traits>;
39 using NodePtr = typename Base::NodePtr;
40 using value_type = typename Traits::value_type;
41
42 /* ----------------- helpers de altura y balanceo ---------------- */
44 static MyNode* avl(NodePtr n) {
45 return static_cast<MyNode*>(n);
46 }
48 static size_t height(NodePtr n) {
49 return n ? avl(n)->m_height : 0;
50 }
52 static void update_height(NodePtr n) {
53 if (n)
54 avl(n)->m_height = 1 + max(height(n->getChild(0)),
55 height(n->getChild(1)));
56 }
58 static ptrdiff_t bf(NodePtr n) {
59 return static_cast<ptrdiff_t>(height(n->getChild(0)))
60 - static_cast<ptrdiff_t>(height(n->getChild(1)));
61 }
62
63 /* ----------------------- rotaciones ----------------------------- */
66 void rotate_right(NodePtr& y_ref) {
67 auto* y = avl(y_ref);
68 auto* x = avl(y->getChild(0));
69 y->setChild(0, x->getChild(1));
70 x->setChild(1, y);
71 update_height(y);
72 update_height(x);
73 y_ref = x;
74 }
76 void rotate_left(NodePtr& x_ref) {
77 auto* x = avl(x_ref);
78 auto* y = avl(x->getChild(1));
79 x->setChild(1, y->getChild(0));
80 y->setChild(0, x);
81 update_height(x);
82 update_height(y);
83 x_ref = y;
84 }
85
86 /* ---------------- rebalanceo post-inserción --------------------- */
93 void rebalance(NodePtr& p) {
94 update_height(p);
95 ptrdiff_t b = bf(p);
96 if (b > 1) {
97 if (bf(p->getChild(0)) < 0)
98 rotate_left(p->getChildRef(0)); // LR → convierte a LL
99 rotate_right(p); // LL
100 } else if (b < -1) {
101 if (bf(p->getChild(1)) > 0)
102 rotate_right(p->getChildRef(1)); // RL → convierte a RR
103 rotate_left(p); // RR
104 }
105 }

109 NodePtr make_node(const value_type& v, Ref ref) override {
110 return new MyNode(v, ref);
111 }

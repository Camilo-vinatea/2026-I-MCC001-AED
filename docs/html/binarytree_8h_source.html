117template <typename Container, typename Policy>
118class BinaryTreeIterator
119 : public general_iterator<Container, BinaryTreeIterator<Container, Policy>>
120{
121 using MySelf = BinaryTreeIterator<Container, Policy>;
122 using Parent = general_iterator<Container, MySelf>;
123public:
124 using Node = typename Container::Node;
125private:
126 deque<Node*> m_cola;
127
129 void avanzar() {
130 if (!m_cola.empty()) { this->m_pNode = m_cola.front(); m_cola.pop_front(); }
131 else { this->m_pNode = nullptr; }
132 }
133public:
135 BinaryTreeIterator(Container* pC, Node* pRaiz)
136 : Parent(pC, nullptr) { Policy::construir(m_cola, pRaiz); avanzar(); }
137
139 MySelf& operator++() { avanzar(); return *this; }
140};

455 void insertar_interno(NodePtr& pNodo, const value_type& value, Ref ref) {
456 if (!pNodo) {
457 pNodo = make_node(value, ref);
458 ++m_size;
459 return;
460 }
461 size_t pos = !m_comp(value, pNodo->getDataRef());
462 insertar_interno(pNodo->getChildRef(pos), value, ref);
463 post_insert(pNodo);
464 }

469 dst = new Node(src->getData(), src->getRef());

471 copiar_interno(dst->getChildRef(1), src->getChild(1));
